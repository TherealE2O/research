🛠️ Integrating into Your Music Player / App

Import the visualizer module and attach it to any HTML5 <audio> element in 3 lines:

import { AudiogramVisualizer } from './audiogram_visualizer.js';

const canvas = document.getElementById('myCanvas');
const audioEl = document.getElementById('myAudio');

const visualizer = new AudiogramVisualizer(canvas, { numBars: 46, barColor: '#ffffff' });
audioEl.onplay = () => visualizer.connectAudioElement(audioEl);
